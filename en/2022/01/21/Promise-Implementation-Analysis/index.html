1 - Promise Basic Model (Synchronous, Execution Flow Analysis)

  1. Promise is a class that receives a constructor function as a parameter. When executing the class, the constructor function is also executed synchronously.
  2. When the constructor function executes synchronously, it passes two callback functions: success callback [resolve] and failure callback [reject].
  • Success callback: receives a parameter in the callback function, which can be either a plain value or a Promise object. For now, we only consider plain values.
  • Failure callback: receives a parameter in the callback function, usually the reason for failure
  1. Promise has three states: pending (waiting), fulfilled (success), and rejected (failure). State transitions are only allowed from pending to fulfilled or from pending to rejected. Once a state is determined, it cannot be changed.
  2. The then method receives two parameters: success callback [successCallback] function and failure callback [failCallback] function. When the then method is executed, it executes the success callback or failure callback based on the Promise’s execution state, and passes the corresponding cached state value to the callback function as a parameter.
// Promise three states
const PENDING = "pending"; // waiting
const FULFILLED = "fulfilled"; // success
const REJECTED = "rejected"; // failure

/**
 *   Simulating native Promise
 *
 *   @class MyPromise
 *   @constructor executor - constructor function that receives two callback functions
 */
class MyPromise {
  constructor(executor) {
    executor(this.resolve, this.reject);
  }

  // Default state is pending
  status = PENDING;

  // Cache state values
  value = undefined; // success state [parameter]
  reason = undefined; // failure state [reason]

  /**
   *   Success callback
   *
   *   @see Using arrow function to keep 'this' pointing to current MyPromise object
   */
  resolve = (value) => {
    // State cannot be changed once determined, return early
    if (this.status !== PENDING) return;

    // Change state to fulfilled
    this.status = FULFILLED;

    // Cache the value passed to success callback
    this.value = value;
  };

  /**
   * Failure callback
   */
  reject = (reason) => {
    // State cannot be changed once determined, return early
    if (this.status !== PENDING) return;

    // Change state to rejected
    this.status = REJECTED;

    // Cache the reason passed to failure callback
    this.reason = reason;
  };

  /**
   *   then method
   *
   *   @params {function} successCallback - success callback function
   *   @params {function} failCallback - failure callback function
   */
  then(successCallback, failCallback) {
    if (this.status === FULFILLED) successCallback(this.value);
    if (this.status === REJECTED) failCallback(this.reason);
  }
}

// Test current MyPromise

let promiseTest = new MyPromise((resolve, reject) => {
  resolve("success");
  // reject('failure')
});
// Execute then method
promiseTest.then(
  (value) => {
    console.log(value);
  },
  (reason) => {
    console.log(reason);
  },
);

2 - Promise Asynchronous Characteristics

  1. Asynchronous characteristics mean that the code does not block execution and specific values or functions are obtained after the asynchronous operation completes.
  2. When executing the then method in a Promise and the Promise object’s state is pending (waiting state), it is determined as an asynchronous state.
  3. When determining an asynchronous state, successCallback (success callback) and failCallback (failure callback) need to be cached in the Promise object.
  4. When the asynchronous method in the constructor completes and calls resolve (success callback) or reject (failure callback), after completing the preceding logic, the successCallback or failCallback mounted to the Promise object through the external then method is executed.

    // Promise three states
    const PENDING = 'pending' // waiting
    const FULFILLED = 'fulfilled' // success
    const REJECTED = 'rejected' // failure

    /**
    *   Simulating native Promise
    *
    *   @class MyPromise
    *   @constructor executor - constructor function that receives two callback functions
    */
     resolve = (value) => {
         // State cannot be changed once determined, return early
         if (this.status !== PENDING) return

         // Change state to fulfilled
         this.status = FULFILLED

         // Cache the value passed to success callback
         this.value = value

         // Execute success callback
         this.successCallback && this.successCallback(this.value)
     }

     /**
     * Failure callback
     */
     reject = (reason) => {
         // State cannot be changed once determined, return early
         if (this.status !== PENDING) return

         // Change state to rejected
         this.status = REJECTED

         // Cache the reason passed to failure callback
         this.reason = reason

         // Execute failure callback
         this.failCallback && this.failCallback(this.reason)
     }

    /**
    *   then method
    *
    *   @params {function} successCallback - success callback function
    *   @params {function} failCallback - failure callback function
    */
    then ( successCallback, failCallback ) {
        if (this.status === FULFILLED) successCallback(this.value)
        else if (this.status === REJECTED) failCallback(this.reason)
        else {
            // Cache success callback
            this.successCallback = successCallback
            // Cache failure callback
            this.failCallback = failCallback
        }
    }
}


// Test current MyPromise

let promiseTest = new MyPromise((resolve, reject) => {
    setTimeout(() => {
        resolve('success')
        // reject('failure')
    }, 2000)
})
promiseTest.then((value) => {
    console.log('Success callback: ', value)
}, (reason) => {
    console.log('Failure reason: ', reason)
})

3 - Promise Multiple Call Support Implementation

  1. The Promise method can be called multiple times, and each then method should be executed correctly after the call.
  2. Refactor the container for caching callback functions to an array, so that each time the then method is executed, successCallback (success callback) and failCallback (failure callback) should enter the corresponding cache stack.
  3. When calling resolve (success callback) or reject (failure callback) in the constructor, loop through and execute the cached callback functions, passing the current Promise’s success data or failure reason to the callback function.
// Promise three states
const PENDING = "pending"; // waiting
const FULFILLED = "fulfilled"; // success
const REJECTED = "rejected"; // failure
/**
 *   Simulating native Promise
 *
 *   @class MyPromise
 *   @constructor executor - constructor function that receives two callback functions
 */
class MyPromise {
  constructor(executor) {
    executor(this.resolve, this.reject);
  }
  // Default state is pending
  status = PENDING;

  // Cache state values
  value = undefined; // success state [parameter]
  reason = undefined; // failure state [reason]
  // Cache callback functions

  successCallback = []; // success callback
  failCallback = []; // failure callback
  /**
   *   Success callback
   *
   *   @see Using arrow function to keep 'this' pointing to current MyPromise object
   */
  resolve = (value) => {
    // State cannot be changed once determined, return early
    if (this.status !== PENDING) return;

    // Change state to fulfilled
    this.status = FULFILLED;

    // Cache the value passed to success callback
    this.value = value;

    // Execute success callbacks
    while (this.successCallback.length)
      this.successCallback.shift()(this.value);
  };
  /**
   * Failure callback
   */
  reject = (reason) => {
    // State cannot be changed once determined, return early
    if (this.status !== PENDING) return;

    // Change state to rejected
    this.status = REJECTED;

    // Cache the reason passed to failure callback
    this.reason = reason;

    // Execute failure callbacks
    while (this.failCallback.length) this.failCallback.shift()(this.reason);
  };
  /**
   *   then method
   *
   *   @params {function} successCallback - success callback function
   *   @params {function} failCallback - failure callback function
   */
  then(successCallback, failCallback) {
    if (this.status === FULFILLED) successCallback(this.value);
    else if (this.status === REJECTED) failCallback(this.reason);
    else {
      // Cache success callback
      this.successCallback.push(successCallback);
      // Cache failure callback
      this.failCallback.push(failCallback);
    }
  }
}

// Test current MyPromise

let promiseTest = (status, num) => {
  return new MyPromise((resolve, reject) => {
    setTimeout(() => {
      if (status) resolve(`Success: ${num}`);
      else reject(`Failure: ${num}`);
    }, 2000);
  });
};
promiseTest(true, 1).then(
  (value) => {
    console.log("Success callback: ", value);
  },
  (reason) => {
    console.log("Failure reason: ", reason);
  },
);
promiseTest(false, 2).then(
  (value) => {
    console.log("Success callback: ", value);
  },
  (reason) => {
    console.log("Failure reason: ", reason);
  },
);
