Getting File Stream from Upload Input

Step 1: Create Input Element

MDN input file documentation

<input type="file" name="file" />
<!-- Attribute values can omit quotes to reduce file size -->

Step 2: Get File Stream

The return value is a FileList collection (array-like):

document.querySelector("input[name=file]").files;

Convert File to Base64

Check if String is Base64

function isBase64(str) {
  if (str === "" || str.trim() === "") {
    return false;
  }
  try {
    return btoa(atob(str)) == str;
  } catch (err) {
    return false;
  }
}

Get File as Base64 String

function getBase64(file) {
  return new Promise((resolve, reject) => {
    try {
      // Check if already Base64 string
      if (
        Object.prototype.toString.call(file) === "[object String]" &&
        isBase64(file)
      ) {
        resolve(file);
      } else {
        // Convert file to Base64
        const reader = new FileReader();
        reader.readAsDataURL(file);
        reader.onload = () => resolve(reader.result);
        reader.onerror = (error) => reject(error);
      }
    } catch (err) {
      reject(err);
    }
  });
}

// Usage
getBase64(fileInputElement.files[0])
  .then((base64) => console.log("Base64:", base64))
  .catch((err) => console.error("Error:", err));

Convert Base64 to Binary/Blob

function base64ToBlob(urlData, type) {
  let arr = urlData.split(",");
  let mime, bytes;

  if (arr && arr.length === 2) {
    // Extract MIME type from data URL
    mime = arr[0].match(/:(.*?);/)[1] || type;
    // Decode Base64 to byte string
    bytes = window.atob(arr[1]);
  } else {
    mime = type;
    bytes = window.atob(arr[0]);
  }

  // Handle edge case: convert ASCII codes < 0 to > 0
  let ab = new ArrayBuffer(bytes.length);
  let ia = new Uint8Array(ab);

  for (let i = 0; i < bytes.length; i++) {
    ia[i] = bytes.charCodeAt(i);
  }

  return new Blob([ab], {
    type: mime,
  });
}

// Usage
const blob = base64ToBlob(base64String, "image/png");

Add Filename to Binary

Method 1: When Uploading to Server

let formData = new FormData();
formData.append("file", base64ToBlob(base64String), "filename.jpg");
// "file" is the form field name sent to backend
// "filename.jpg" is the file name

Method 2: Create File Object Without Upload

const file = new File([base64ToBlob(base64String)], "filename.jpg");
// "filename.jpg" is the file name to set

Upload File to Server

function handleUpload() {
  let formData = new FormData();
  formData.append(
    "file",
    document.querySelector("input[name=file]").files[0],
    "filename.jpg",
  );

  let xhr = new XMLHttpRequest();

  // Set request method and URL
  xhr.open("POST", "/api/upload", true);

  // Set up event handlers
  xhr.onload = uploadComplete;
  xhr.onerror = uploadFailed;
  xhr.upload.onprogress = progressFunction;

  // Important: Don't set Content-Type header manually
  // Browser will set it to multipart/form-data automatically

  // Send the request
  xhr.send(formData);
}

// Handle upload completion
function uploadComplete(event) {
  if (event.target.status === 200) {
    console.log("Upload successful!");
    console.log(event.target.responseText);
  }
}

// Handle upload errors
function uploadFailed(event) {
  console.error("Upload failed:", event.target.statusText);
}

// Track upload progress
function progressFunction(event) {
  if (event.lengthComputable) {
    let percentComplete = (event.loaded / event.total) * 100;
    console.log("Upload progress: " + percentComplete + "%");
  }
}

Complete Upload Example with Modern Fetch API

async function uploadFile() {
  const fileInput = document.querySelector("input[name=file]");
  const file = fileInput.files[0];

  if (!file) {
    console.error("No file selected");
    return;
  }

  const formData = new FormData();
  formData.append("file", file);

  try {
    const response = await fetch("/api/upload", {
      method: "POST",
      body: formData,
      // Don't set Content-Type header
    });

    if (response.ok) {
      const result = await response.json();
      console.log("Upload successful:", result);
    } else {
      console.error("Upload failed:", response.statusText);
    }
  } catch (error) {
    console.error("Upload error:", error);
  }
}

Best Practices

  1. Always validate file type - Check MIME type on both frontend and backend
  2. Limit file size - Prevent large file uploads
  3. Use Content-Type: multipart/form-data - Don’t set manually; let browser handle it
  4. Handle errors gracefully - Show user-friendly error messages
  5. Show progress - Use upload progress events
  6. Secure backend - Validate and sanitize files on server

Common Issues and Solutions

Issue Solution
File size too large Use FormData with chunking
CORS error Configure server CORS
Timeout Increase timeout or use chunks
Memory issues with Base64 Stream directly to FormData instead
Wrong MIME type Check file extension and content

Browser Compatibility

  • ✅ FileReader API - IE10+
  • ✅ Blob API - IE10+
  • ✅ FormData API - IE10+
  • ✅ Fetch API - Most modern browsers
  • ✅ XMLHttpRequest - All browsers
